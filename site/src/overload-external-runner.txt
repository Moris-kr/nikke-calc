# python run-battle.py job.json module-result.json result.json
# Only the calculator's public, version-checked engine is downloaded. Inputs are data, never evaluated.
import json, pathlib, sys, tempfile, urllib.request, zipfile
job=json.loads(pathlib.Path(sys.argv[1]).read_text(encoding='utf-8'))
result=json.loads(pathlib.Path(sys.argv[2]).read_text(encoding='utf-8'))
if result['requestId'] != job['requestId']: raise ValueError('requestId mismatch')
version=job['engineVersion']
if not isinstance(version,str) or len(version)!=16 or any(c not in '0123456789abcdef' for c in version): raise ValueError('Invalid version')
with tempfile.TemporaryDirectory() as folder:
 root=pathlib.Path(folder)
 # If network is unavailable, place the downloaded versioned ZIP beside this script.
 filename='external-engine-'+version+'.zip'
 local=pathlib.Path(filename)
 archive=local if local.is_file() else root/filename
 if not local.is_file(): urllib.request.urlretrieve('https://moris-kr.github.io/nikke-calc/'+filename,archive)
 with zipfile.ZipFile(archive) as z:
  for member in z.infolist():
   destination=(root/member.filename).resolve()
   if not destination.is_relative_to(root.resolve()): raise ValueError('Invalid archive path')
  z.extractall(root)
 if json.loads((root/'manifest.json').read_text(encoding='utf-8'))['version']!=version: raise ValueError('Engine version mismatch')
 sys.path.insert(0,str(root))
 import bridge
 outputs=[]
 for item in job['simulations']:
  request=json.loads(json.dumps(item['request']))
  for character in request.get('characters',{}).values(): character.pop('overloadLines',None)
  raw=json.loads(bridge.run_request(json.dumps(request,ensure_ascii=False)))
  fields=['squadTotal','duration','hitCount','charTotals','deviations','previewNote']
  outputs.append({'id':item['id'],'result':{key:raw[key] for key in fields if key in raw}})
 result['engineVersion']=version
 result['simulations']=outputs
 pathlib.Path(sys.argv[3]).write_text(json.dumps(result,ensure_ascii=False),encoding='utf-8')
